     if (!featureFlags.metaculusDivergence) {
       skipped.push({
         detector: 'metaculus-divergence',
         reason: 'Feature flag metaculusDivergence is disabled (requires Phase 4)',

     } else {
       try {
         const allMarkets = [...polymarketMarkets, ...kalshiMarkets];
         const metaculusOpps = await this.metaculusDetector.detect(allMarkets);

     private normalizeMetaculusDivergence(opp: MetaculusDivergenceOpportunity): UnifiedOpportunity {

         platform: opp.marketPlatform,
         marketId: opp.marketId,

         grossEdge: opp.divergencePercent / 100,

  <files>src/dashboard/components/MetaculusView.tsx</files>
  <action>
Create React component for displaying Metaculus divergence details in the dashboard.

Follow existing patterns from SettlementView.tsx.

Component structure:
```tsx
import React from 'react';
import { Box, Text } from 'ink';

interface MetaculusViewProps {
  metaculusId: number;
  metaculusTitle: string;
  metaculusPrediction: number;
  marketPrice: number;
  divergencePercent: number;
  forecastTimestamp: string;
  forecastAge: number;
  isFresh: boolean;
  stalenessWarning?: string;
  matchConfidence: number;
}

export function MetaculusView({
  metaculusId,
  metaculusTitle,
  metaculusPrediction,
  marketPrice,
  divergencePercent,
  forecastTimestamp,
  forecastAge,
  isFresh,
  stalenessWarning,
  matchConfidence,
}: MetaculusViewProps): React.JSX.Element {
  // Color coding for staleness
  const stalenessColor = isFresh ? 'green' : forecastAge > 14 ? 'red' : 'yellow';

  return (
    <Box flexDirection="column" marginTop={1}>
      <Text bold>Metaculus Divergence</Text>
      <Box marginLeft={2} flexDirection="column">
        <Text>
          <Text color="cyan">Question: </Text>
          <Text>{metaculusTitle}</Text>
        </Text>
        <Text>
          <Text color="cyan">Metaculus ID: </Text>
          <Text>{metaculusId}</Text>
        </Text>
        <Box marginTop={1}>
          <Text>
            <Text color="cyan">Metaculus: </Text>
            <Text bold>{(metaculusPrediction * 100).toFixed(1)}%</Text>
            <Text>  |  </Text>
            <Text color="cyan">Market: </Text>
            <Text bold>{(marketPrice * 100).toFixed(1)}%</Text>
            <Text>  |  </Text>
            <Text color="cyan">Divergence: </Text>
            <Text bold color="yellow">{divergencePercent.toFixed(1)}%</Text>
          </Text>
        </Box>
        <Box marginTop={1}>
          <Text>
            <Text color="cyan">Forecast Age: </Text>
            <Text color={stalenessColor}>{forecastAge} days</Text>
            {!isFresh && stalenessWarning && (
              <Text color="yellow"> ({stalenessWarning})</Text>
            )}
          </Text>
        </Box>
        <Text>
          <Text color="cyan">Match Confidence: </Text>
          <Text>{(matchConfidence * 100).toFixed(0)}%</Text>
        </Text>
        <Text dimColor>
          Last updated: {new Date(forecastTimestamp).toLocaleDateString()}
        </Text>
      </Box>
    </Box>
  );
}
```
