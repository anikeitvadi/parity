```tsx
import React from 'react';
import { Box, Text } from 'ink';
import { Select } from '@inkjs/ui';
import { ScoredOpportunity } from '../../scoring/types.js';

interface Props {
  opportunities: ScoredOpportunity[];
  onSelect: (id: string) => void;
}

export function OpportunityTable({ opportunities, onSelect }: Props) {
  if (opportunities.length === 0) {
    return <Text color="yellow">No opportunities above threshold</Text>;
  }

  const options = opportunities.map(opp => ({
    label: formatRow(opp),
    value: opp.id,
  }));

  return (
    <Box flexDirection="column">
      <Box marginBottom={1}>
        <Text bold>{'SCORE'.padEnd(7)}</Text>
        <Text bold>{'MARKET'.padEnd(42)}</Text>
        <Text bold>{'EDGE'.padStart(8)}</Text>
        <Text bold>{'TYPE'.padStart(14)}</Text>
        <Text bold>{'SIZE'.padStart(10)}</Text>
      </Box>
      <Select options={options} onChange={onSelect} />
    </Box>
  );
}

function formatRow(opp: ScoredOpportunity): string {
  const score = formatScore(opp.score);
  const market = opp.marketQuestion.slice(0, 40).padEnd(40);
  const edge = ((opp.netEdge * 100).toFixed(1) + '%').padStart(7);
  const type = opp.type.slice(0, 12).padStart(12);
  const size = ('$' + opp.positionSize.toFixed(0)).padStart(8);
  return `${score} ${market} ${edge} ${type} ${size}`;
}

function formatScore(score: number): string {
  const text = score.toFixed(1).padStart(5);
  // Note: Ink Select doesn't support inline colors in labels
  // Color coding handled by parent component
  return text;
}
```

```tsx
import React from 'react';
import { Box, Text, Newline } from 'ink';
import { ScoredOpportunity } from '../../scoring/types.js';

interface Props {
  opportunity: ScoredOpportunity;
  onBack: () => void;
}

export function OpportunityDetail({ opportunity, onBack }: Props) {
  const breakdown = opportunity.scoreBreakdown;

  return (
    <Box flexDirection="column" paddingX={1}>
      <Text bold underline>{opportunity.marketQuestion}</Text>
      <Newline />

      <Box>
        <Text color="cyan">Score: </Text>
        <ScoreText score={opportunity.score} />
        <Text dimColor> (edge:{breakdown.edgeScore.toFixed(1)} conf:{breakdown.confidenceScore.toFixed(1)} liq:{breakdown.liquidityScore.toFixed(1)})</Text>
      </Box>

      <Box>
        <Text color="cyan">Edge: </Text>
        <Text>{(opportunity.netEdge * 100).toFixed(1)}% net</Text>
        <Text dimColor> ({(opportunity.grossEdge * 100).toFixed(1)}% gross)</Text>
      </Box>

      <Box>
        <Text color="cyan">Type: </Text>
        <Text>{opportunity.type}</Text>
        <Text dimColor> on {opportunity.platform}</Text>
      </Box>

      <Box>
        <Text color="cyan">Liquidity: </Text>
        <Text>${opportunity.minLiquidity.toLocaleString()}</Text>
      </Box>

      <Box>
        <Text color="cyan">Position: </Text>
        <Text bold>${opportunity.positionSize.toFixed(0)}</Text>
        <Text> ({opportunity.positionPercent.toFixed(1)}% of bankroll)</Text>
      </Box>

      {opportunity.closeDate && (
        <Box>
          <Text color="cyan">Closes: </Text>
          <Text>{opportunity.closeDate}</Text>
        </Box>
      )}

      <Newline />
      <Text dimColor>Market ID: {opportunity.marketId}</Text>
      <Text dimColor>Detected: {new Date(opportunity.detectedAt).toLocaleString()}</Text>

      <Newline />
      <Text dimColor>Press 'b' or Escape to go back</Text>
    </Box>
  );
}

function ScoreText({ score }: { score: number }) {
  if (score >= 7) return <Text color="green" bold>{score.toFixed(1)}</Text>;
  if (score >= 5) return <Text color="yellow">{score.toFixed(1)}</Text>;
  return <Text dimColor>{score.toFixed(1)}</Text>;
}
```

Create StatusBar.tsx:
```tsx
import React from 'react';
import { Box, Text } from 'ink';

interface Props {
  lastUpdate: Date | null;
  opportunityCount: number;
  errorCount: number;
  isLoading: boolean;
}

export function StatusBar({ lastUpdate, opportunityCount, errorCount, isLoading }: Props) {
  const staleMinutes = lastUpdate
    ? Math.floor((Date.now() - lastUpdate.getTime()) / 60000)
    : null;
  const isStale = staleMinutes !== null && staleMinutes > 30;

  return (
    <Box borderStyle="single" paddingX={1}>
      <Text>
        {isLoading ? '⟳ Refreshing...' : `✓ ${opportunityCount} opportunities`}
      </Text>
      <Text> | </Text>
      {lastUpdate ? (
        <Text color={isStale ? 'yellow' : undefined}>
          Updated {staleMinutes}m ago
        </Text>
      ) : (
        <Text dimColor>Never updated</Text>
      )}
      {errorCount > 0 && (
        <>
          <Text> | </Text>
          <Text color="red">{errorCount} errors</Text>
        </>
      )}
    </Box>
  );
}
```

Create App.tsx (main dashboard):
```tsx
import React, { useState, useEffect } from 'react';
import { Box, Text, useInput, useApp } from 'ink';
import { Spinner } from '@inkjs/ui';
import { OpportunityTable } from './components/OpportunityTable.js';
import { OpportunityDetail } from './components/OpportunityDetail.js';
import { StatusBar } from './components/StatusBar.js';
import { OpportunityAggregator } from '../aggregator/opportunity-aggregator.js';
import { OpportunityDeduplicator } from '../aggregator/deduplicator.js';
import { scoreOpportunity, ScoredOpportunity } from '../scoring/index.js';

interface Props {
  bankroll: number;
  minScore: number;
  watchMode: boolean;
  refreshInterval: number;
}

export function App({ bankroll, minScore, watchMode, refreshInterval }: Props) {
  const { exit } = useApp();
  const [opportunities, setOpportunities] = useState<ScoredOpportunity[]>([]);
  const [selectedId, setSelectedId] = useState<string | null>(null);
  const [lastUpdate, setLastUpdate] = useState<Date | null>(null);
  const [errorCount, setErrorCount] = useState(0);
  const [isLoading, setIsLoading] = useState(true);

  const aggregator = React.useRef(new OpportunityAggregator());
  const deduplicator = React.useRef(new OpportunityDeduplicator());

  const refresh = React.useCallback(async () => {
    setIsLoading(true);
    try {
      const result = await aggregator.current.aggregate();
      setErrorCount(result.errors.length);

      // Score and filter
      const scored: ScoredOpportunity[] = [];
      for (const opp of result.opportunities) {
        const scoredOpp = scoreOpportunity(opp, bankroll);
        if (scoredOpp && scoredOpp.score >= minScore) {
          if (!deduplicator.current.isDuplicate(opp)) {
            deduplicator.current.record(opp, scoredOpp.score);
            scored.push(scoredOpp);
          }
        }
      }

      // Sort by score descending
      scored.sort((a, b) => b.score - a.score);
      setOpportunities(scored);
      setLastUpdate(new Date());
    } finally {
      setIsLoading(false);
    }
  }, [bankroll, minScore]);

  // Initial load
  useEffect(() => {
    refresh();
  }, [refresh]);

  // Watch mode polling
  useEffect(() => {
    if (!watchMode) return;
    const timer = setInterval(refresh, refreshInterval);
    return () => clearInterval(timer);
  }, [watchMode, refreshInterval, refresh]);

  // Keyboard navigation
  useInput((input, key) => {
    if (input === 'q' || (key.ctrl && input === 'c')) {
      exit();
    }
    if (selectedId && (input === 'b' || key.escape)) {
      setSelectedId(null);
    }
    if (input === 'r') {
      refresh();
    }
  });

  const selectedOpp = selectedId
    ? opportunities.find(o => o.id === selectedId)
    : null;

  return (
    <Box flexDirection="column">
      <Box marginBottom={1}>
        <Text bold color="cyan">PREDICTION MARKET EDGE SCANNER</Text>
        {watchMode && <Text dimColor> (watching)</Text>}
      </Box>

      {isLoading && !lastUpdate ? (
        <Spinner label="Loading opportunities..." />
      ) : selectedOpp ? (
        <OpportunityDetail
          opportunity={selectedOpp}
          onBack={() => setSelectedId(null)}
        />
      ) : (
        <OpportunityTable
          opportunities={opportunities}
          onSelect={setSelectedId}
        />
      )}

      <Box marginTop={1}>
        <StatusBar
          lastUpdate={lastUpdate}
          opportunityCount={opportunities.length}
          errorCount={errorCount}
          isLoading={isLoading}
        />
      </Box>

      <Text dimColor>q: quit | r: refresh | Enter: details | b: back</Text>
    </Box>
  );
}
```

```typescript
#!/usr/bin/env node
import React from 'react';
import { render } from 'ink';
import meow from 'meow';
import { App } from './dashboard/index.js';

const cli = meow(`
  Usage
    $ npm run dashboard

  Options
    --bankroll, -b  Your total capital (default: 500)
    --min-score, -m Minimum score to display (default: 0)
    --watch, -w     Enable watch mode with auto-refresh
    --interval, -i  Refresh interval in seconds (default: 300)

  Examples
    $ npm run dashboard -- --bankroll 1000 --watch
    $ npm run dashboard -- -b 500 -m 5 -w -i 60
`, {
  importMeta: import.meta,
  flags: {
    bankroll: { type: 'number', shortFlag: 'b', default: 500 },
    minScore: { type: 'number', shortFlag: 'm', default: 0 },
    watch: { type: 'boolean', shortFlag: 'w', default: false },
    interval: { type: 'number', shortFlag: 'i', default: 300 },
  },
});

render(
  <App
    bankroll={cli.flags.bankroll}
    minScore={cli.flags.minScore}
    watchMode={cli.flags.watch}
    refreshInterval={cli.flags.interval * 1000}
  />
);
```
